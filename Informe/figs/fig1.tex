\newcommand{\pipelineimage}[3]{%
    \IfFileExists{#3}{%
        \includegraphics[width=#1,height=#2,keepaspectratio]{#3}%
    }{%
        \fbox{%
            \begin{minipage}[c][#2][c]{#1}

                \scriptsize
                \texttt{\detokenize{#3}}\\[-1pt]

            \end{minipage}%

    }%
}
